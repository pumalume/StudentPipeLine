CREATE TABLE IF NOT EXISTS "Students" (
    student_id SERIAL PRIMARY KEY,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    phone VARCHAR(20),
    date_of_birth DATE,
    enrollment_date DATE DEFAULT CURRENT_DATE,
    status VARCHAR(20) DEFAULT 'active'
);

-- Documentation comments stored in PostgreSQL
COMMENT ON TABLE "Students" IS 'Stores student personal profiles and enrollment status.';
COMMENT ON COLUMN "Students".student_id IS 'Primary key auto-incrementing integer ID.';
COMMENT ON COLUMN "Students".email IS 'Unique contact and portal login email address.';
